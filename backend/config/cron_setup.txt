# Configuración de Cron Job para OpenRoadCyL
# Actualización automática de datos de incidencias

# ============================================
# INSTRUCCIONES DE INSTALACIÓN
# ============================================

# 1. Abrir el crontab del usuario:
#    crontab -e

# 2. Agregar una de las siguientes líneas según la frecuencia deseada:

# --- OPCIONES DE FRECUENCIA ---

# Cada 15 minutos (recomendado para datos en tiempo real):
*/15 * * * * /usr/bin/php /ruta/completa/al/proyecto/backend/scripts/update_data.php >> /ruta/completa/al/proyecto/backend/logs/cron.log 2>&1

# Cada 30 minutos:
*/30 * * * * /usr/bin/php /ruta/completa/al/proyecto/backend/scripts/update_data.php >> /ruta/completa/al/proyecto/backend/logs/cron.log 2>&1

# Cada hora:
0 * * * * /usr/bin/php /ruta/completa/al/proyecto/backend/scripts/update_data.php >> /ruta/completa/al/proyecto/backend/logs/cron.log 2>&1

# Cada 2 horas:
0 */2 * * * /usr/bin/php /ruta/completa/al/proyecto/backend/scripts/update_data.php >> /ruta/completa/al/proyecto/backend/logs/cron.log 2>&1

# Solo en horario laboral (cada 30 min de 8:00 a 20:00):
*/30 8-20 * * * /usr/bin/php /ruta/completa/al/proyecto/backend/scripts/update_data.php >> /ruta/completa/al/proyecto/backend/logs/cron.log 2>&1
